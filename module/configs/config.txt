# KEY=VALUE toggles, one per line, blank = skip that toggle
#
# hide_sus_mnts_for_non_su_procs: 0 = don't hide, 1 = hide all sus mounts
#   for non-su processes (default 0 in kernel). The early value protects zygote
#   startup from caching Sus mounts. The late value is deliberately 0 so normal
#   app and isolated-process mount views are not blanket-filtered after boot;
#   targeted KernelSU umount and Sus path rules remain active.
#
#   Keep HIDE_SUS_MNTS_LATE=1 only for a stack that explicitly requires the
#   legacy always-on behavior. It can create cross-process mount-table drift.
#
# enable_log: 0 = disable susfs kernel log, 1 = enable
#
# enable_avc_log_spoofing: 0 = disable, 1 = spoof the sus 'su' tcontext as
#   'u:r:priv_app:s0:c512,c768' in avc log, bypassing 'su' domain detection
#   via /proc/<pid> enumeration (default 0 in kernel). Disable this when
#   debugging permission/selinux issues.
#
HIDE_SUS_MNTS_NON_SU=1
HIDE_SUS_MNTS_LATE=0
ENABLE_LOG=0
ENABLE_AVC_LOG_SPOOFING=1

# KernelSU kernel_umount feature policy:
#   unchanged = do not change the current KernelSU feature state
#   enabled   = enable by the stable feature name (default)
#   disabled  = disable it and do not register mountpoints
KERNEL_UMOUNT_MODE=enabled

# Automatically register only mountinfo entries identified by source=KSU or
# module-backed root/source/overlay data. Explicit entries remain available in
# kernel_umount.txt. The global KernelSU umount list is never wiped.
AUTO_KERNEL_UMOUNT=1

# Whole-partition/root mountpoints create different views between ordinary and
# isolated app processes under KernelSU. Preserve them in kernel_umount.txt for
# review, but do not register them unless this compatibility override is 1.
ALLOW_BROAD_KERNEL_UMOUNT=0

# Developer Options / ADB handling:
#   unchanged        = never touch settings, USB properties, or adbd (default)
#   spoof-off        = refuse safely until the active stack can spoof Settings
#                      provider reads without changing the real ADB transport
#   actually-disable = write all three settings to 0 and stop/verify adbd
ADB_MODE=unchanged

# Required only for ADB_MODE=actually-disable. This explicit token prevents a
# restored or mistyped config from silently cutting off debugging access.
ADB_DISABLE_CONFIRM=
